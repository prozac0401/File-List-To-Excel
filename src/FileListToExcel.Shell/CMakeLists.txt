cmake_minimum_required(VERSION 3.21)
project(FileListToExcelShell VERSION 1.1.0 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "File List to Excel shell integration requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(FileListToExcel.Shell SHARED ShellExtension.cpp ShellExtension.def Version.rc)
set_target_properties(FileListToExcel.Shell PROPERTIES PREFIX "" OUTPUT_NAME "FileListToExcel.Shell")
target_compile_definitions(FileListToExcel.Shell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(FileListToExcel.Shell PRIVATE shell32 ole32 advapi32 user32 uuid)
if(MSVC)
  target_compile_options(FileListToExcel.Shell PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /guard:cf)
  target_link_options(FileListToExcel.Shell PRIVATE /DYNAMICBASE /NXCOMPAT /GUARD:CF)
else()
  target_compile_options(FileListToExcel.Shell PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
  target_link_options(FileListToExcel.Shell PRIVATE -static -Wl,--dynamicbase,--nxcompat)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(ShellTests tests/ShellTests.cpp)
  target_compile_definitions(ShellTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(ShellTests PRIVATE shell32 ole32 user32 uuid)
  if(MSVC)
    target_compile_options(ShellTests PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(ShellTests PRIVATE -Wall -Wextra -Werror)
    target_link_options(ShellTests PRIVATE -static -municode)
  endif()
  add_executable(RequestSink tests/RequestSink.cpp)
  set_target_properties(RequestSink PROPERTIES OUTPUT_NAME "FileListToExcel")
  target_compile_definitions(RequestSink PRIVATE UNICODE _UNICODE NOMINMAX)
  if(NOT MSVC)
    target_link_options(RequestSink PRIVATE -static -municode)
  endif()
  add_dependencies(ShellTests FileListToExcel.Shell RequestSink)
  add_test(NAME ShellIntegration COMMAND ShellTests $<TARGET_FILE:FileListToExcel.Shell> --dispatch)
endif()
