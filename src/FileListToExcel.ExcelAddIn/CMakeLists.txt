cmake_minimum_required(VERSION 3.21)
project(FileListToExcelExcelAddIn VERSION 1.2.0 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Excel add-in requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(FileListToExcel.ExcelAddIn SHARED ExcelAddIn.cpp ExcelAddIn.def)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
 set_target_properties(FileListToExcel.ExcelAddIn PROPERTIES PREFIX "" OUTPUT_NAME "FileListToExcel.ExcelAddIn.x64")
else()
 set_target_properties(FileListToExcel.ExcelAddIn PROPERTIES PREFIX "" OUTPUT_NAME "FileListToExcel.ExcelAddIn.x86")
endif()
target_compile_definitions(FileListToExcel.ExcelAddIn PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(FileListToExcel.ExcelAddIn PRIVATE ole32 oleaut32 shell32 advapi32 user32 uuid)
if(MSVC)
 target_compile_options(FileListToExcel.ExcelAddIn PRIVATE /W4 /WX /permissive- /utf-8 /EHsc /guard:cf)
 target_link_options(FileListToExcel.ExcelAddIn PRIVATE /DYNAMICBASE /NXCOMPAT /GUARD:CF)
else()
 target_compile_options(FileListToExcel.ExcelAddIn PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
 target_link_options(FileListToExcel.ExcelAddIn PRIVATE -static -Wl,--dynamicbase,--nxcompat)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(ExcelAddInTests tests/ExcelAddInTests.cpp)
 target_compile_definitions(ExcelAddInTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_link_libraries(ExcelAddInTests PRIVATE ole32 oleaut32 user32 uuid)
 if(MSVC)
  target_compile_options(ExcelAddInTests PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
 else()
  target_compile_options(ExcelAddInTests PRIVATE -Wall -Wextra -Werror)
  target_link_options(ExcelAddInTests PRIVATE -static -municode)
 endif()
 add_test(NAME ExcelAddInNative COMMAND ExcelAddInTests $<TARGET_FILE:FileListToExcel.ExcelAddIn>)
endif()
